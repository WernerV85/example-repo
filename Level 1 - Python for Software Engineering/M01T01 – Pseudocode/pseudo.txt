create variable named "highest_num"

request any positive integer via input from user
if first input is greater than zero store integer in variable "highest_num"
	while the user input integer is greater than zero
		compare newly input integer with variable "highest_num"
		if input integer is greater that variable "highest_num"
			replace variable "highest_num" with new integer

	when user input integer is zero
